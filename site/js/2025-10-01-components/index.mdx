```tsx {4-6, 24-26} showLineNumbers
import clsx from 'clsx';

export type GridProps = React.HTMLAttributes<HTMLDivElement> & {
  sm?: number;
  md?: number;
  lg?: number;
};

const Grid = async ({
  children,
  sm = 2,
  md = 2,
  lg = 2,
  className = '',
  ...props
}: GridProps) => {
  return (
    <div
      {...props}
      className={clsx('grid', className)}
      style={
        {
          ...props.style,
          '--columns': sm,
          '--columns-md': md,
          '--columns-lg': lg,
        } as React.CSSProperties
      }
    >
      {children}
    </div>
  );
};

export default Grid;
```

## Spoiler

<Spoiler>
<Code title="spoiler.tsx">
```tsx
import clsx from 'clsx';
import { useId } from 'react';

export type SpoilerProps = React.HTMLAttributes<HTMLDivElement> & {
  hide?: string;
  show?: string;
  expanded?: boolean;
};

const Spoiler = async ({
  children,
  hide = 'Hide',
  show = 'Show',
  expanded = false,
  ...props
}: SpoilerProps) => {
  const id = useId();

  return (
    <div className="spoiler" {...props}>
      <div
        className={clsx('spoiler__content-wrapper', {
          'spoiler__content-wrapper--expanded': expanded,
        })}
        id={id}
      >
        <div className="spoiler__content">{children}</div>
      </div>

      <button
        className="spoiler__toggle btn btn--sm"
        aria-expanded={expanded}
        aria-controls={id}
      >
        <span className="spoiler__toggle-show">{show}</span>
        <span className="spoiler__toggle-hide">{hide}</span>
      </button>
    </div>
  );
};

export default Spoiler;
```
